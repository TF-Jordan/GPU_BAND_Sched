\makeglossaries

%%% Glossaire %%%
\newglossaryentry{paging-distant}{
    name={Paging distant},
    description={Mécanisme de déplacement des pages mémoire froides d'une VM vers la RAM d'un autre nœud du cluster, avec rappel transparent via \texttt{userfaultfd}.}
}
\newglossaryentry{eviction}{
    name=Éviction,
    description={Processus de sélection et d'envoi de pages mémoire froides vers un store distant pour libérer de la RAM locale.}
}
\newglossaryentry{noeud-store}{
    name={Nœud store},
    description={Nœud du cluster dont le rôle principal est de recevoir et stocker en RAM les pages distantes envoyées par d'autres nœuds.}
}
\newglossaryentry{hotplug}{
    name=Hotplug,
    description={Ajout ou retrait dynamique d'un processeur virtuel (vCPU) à une VM en cours d'exécution, sans la redémarrer, via le protocole QMP.}
}
\newglossaryentry{balloon}{
    name=Balloon virtio,
    description={Dispositif virtio permettant d'ajuster dynamiquement la quantité de RAM physique visible par un guest, en gonflant ou dégonflant un ballon logiciel.}
}
